Código Firmware ESP32 (Arduino IDE · C++)

Envía tramas JSON compatibles por USB Serial (115200 baud) y habilita servidor WiFi HTTP (/telemetry) simultáneamente. 

/*
 * PROYECTO: Certeza Bus IoT SMART - Recorrido Casablanca - Valparaíso (Mira Sur)
 * ASIGNATURA: EII 5624 - Tecnologías de la Información: Aplicaciones y Tendencias (PUCV)
 * HARDWARE: Placa ESP32 + Módulo GPS NEO-6M (UART2) + Sensores Infrarrojos Entrada/Salida
 */

#include <WiFi.h>
#include <WebServer.h>
#include <TinyGPS++.h>

// --- CONFIGURACIÓN RED WIFI ---
const char* ssid = "TU_RED_WIFI";
const char* password = "TU_PASSWORD_WIFI";

// --- PINES HARDWARE ESP32 ---
#define PIN_SENSOR_ENTRADA 32  // Sensor óptico/ultrasónico puerta delantera (subida)
#define PIN_SENSOR_SALIDA  33  // Sensor óptico/ultrasónico puerta trasera (bajada)
#define RXD2 16                // GPS TX -> ESP32 RX2
#define TXD2 17                // GPS RX -> ESP32 TX2

// --- VARIABLES DEL BUS ---
const String BUS_ID = "123";
const String PATENTE = "GY-PT-68";
const int CAPACIDAD_MAX = 40;
int aforoActual = 18; // Valor inicial de prueba

// Coordenadas por defecto (Plaza de Casablanca) si el GPS aún no fija satélites
double latitud = -33.3188;
double longitud = -71.4078;

TinyGPSPlus gps;
HardwareSerial gpsSerial(2);
WebServer server(80);

unsigned long ultimoEnvio = 0;
const unsigned long INTERVALO_MS = 1000; // Enviar cada 1 segundo (Latencia < 3s)

// Construye la trama JSON estándar esperada por la interfaz web
String construirJSON() {
  String json = "{";
  json += "\"bus_id\":\"" + BUS_ID + "\",";
  json += "\"patente\":\"" + PATENTE + "\",";
  json += "\"lat\":" + String(latitud, 6) + ",";
  json += "\"lng\":" + String(longitud, 6) + ",";
  json += "\"aforo\":" + String(aforoActual) + ",";
  json += "\"capacidad\":" + String(CAPACIDAD_MAX);
  json += "}";
  return json;
}

// Endpoint HTTP GET /telemetry con cabeceras CORS habilitadas
void handleTelemetry() {
  server.sendHeader("Access-Control-Allow-Origin", "*");
  server.send(200, "application/json", construirJSON());
}

void setup() {
  Serial.begin(115200);
  gpsSerial.begin(9600, SERIAL_8N1, RXD2, TXD2);

  pinMode(PIN_SENSOR_ENTRADA, INPUT_PULLUP);
  pinMode(PIN_SENSOR_SALIDA, INPUT_PULLUP);

  // Conexión WiFi opcional para modo HTTP Polling
  WiFi.begin(ssid, password);
  int intentos = 0;
  while (WiFi.status() != WL_CONNECTED && intentos < 15) {
    delay(500);
    intentos++;
  }

  if (WiFi.status() == WL_CONNECTED) {
    server.on("/telemetry", handleTelemetry);
    server.begin();
  }
}

void loop() {
  // 1. Leer datos NMEA del módulo GPS
  while (gpsSerial.available() > 0) {
    if (gps.encode(gpsSerial.read())) {
      if (gps.location.isValid()) {
        latitud = gps.location.lat();
        longitud = gps.location.lng();
      }
    }
  }

  // 2. Detección de flujo de pasajeros en puertas (flanco activo bajo)
  if (digitalRead(PIN_SENSOR_ENTRADA) == LOW) {
    if (aforoActual < CAPACIDAD_MAX) aforoActual++;
    delay(350); // Antirrebote físico
  }
  if (digitalRead(PIN_SENSOR_SALIDA) == LOW) {
    if (aforoActual > 0) aforoActual--;
    delay(350); // Antirrebote físico
  }

  // 3. Escuchar comandos desde la interfaz web por USB Serial
  if (Serial.available() > 0) {
    String cmd = Serial.readStringUntil('\n');
    cmd.trim();
    if (cmd == "RESET_AFORO") aforoActual = 0;
  }

  // 4. Atender peticiones WiFi HTTP
  if (WiFi.status() == WL_CONNECTED) {
    server.handleClient();
  }

  // 5. Transmitir trama JSON por puerto Serial cada 1 segundo
  if (millis() - ultimoEnvio >= INTERVALO_MS) {
    ultimoEnvio = millis();
    Serial.println(construirJSON());
  }
}
